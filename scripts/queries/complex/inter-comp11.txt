// Given a start Person with ID $personId, find that Person’s friends and friends of friends (excluding start Person) who started working in some Company in a given Country with name $countryName, before a given date ($workFromYear).

/*
personId1: 8796093023522
countryName11: "Hungary"
workFromYear11: 2011
*/
MATCH (person:Person {id: $personId1})-[:knows*1..2]-(friend:Person)
WHERE not(person=friend)
WITH DISTINCT friend
MATCH (friend)-[workAt:workAt]->(company:Organisation)-[:isLocatedIn]->(:Place {name: $countryName11})
WHERE workAt.workFrom < $workFromYear11
RETURN
        friend.id AS personId,
        friend.firstName AS personFirstName,
        friend.lastName AS personLastName,
        company.name AS organizationName,
        workAt.workFrom AS organizationWorkFromYear
ORDER BY
        organizationWorkFromYear ASC,
        toInteger(personId) ASC,
        organizationName DESC
LIMIT 10
